    <script>
        // --- Background Particles Engine ---
        const canvas = document.getElementById('bg-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function initCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }

        class Particle {
            constructor() {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.size = Math.random() * 1.5;
                this.speedX = Math.random() * 0.5 - 0.25;
                this.speedY = Math.random() * 0.5 - 0.25;
                this.opacity = Math.random() * 0.5;
            }
            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                if (this.x > canvas.width) this.x = 0;
                if (this.x < 0) this.x = canvas.width;
                if (this.y > canvas.height) this.y = 0;
                if (this.y < 0) this.y = canvas.height;
            }
            draw() {
                ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        function createParticles() {
            for (let i = 0; i < 150; i++) {
                particles.push(new Particle());
            }
        }

        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            particles.forEach(p => {
                p.update();
                p.draw();
            });
            requestAnimationFrame(animate);
        }

        window.addEventListener('resize', initCanvas);
        initCanvas();
        createParticles();
        animate();

        // --- Custom Cursor ---
        const cursor = document.getElementById('cursor');
        window.addEventListener('mousemove', (e) => {
            gsap.to(cursor, {
                x: e.clientX - 10,
                y: e.clientY - 10,
                duration: 0.1
            });
        });

        // --- Journey Logic ---
        const narrativeSteps = [
            "In a world of 8 billion people...",
            "There are only a handful who truly matter.",
            "But there is only one...",
            "Who makes everything feel like magic.",
            "Today is all about that one person."
        ];

        function startJourney() {
            // Hide Screen 1
            gsap.to("#screen-1", { opacity: 0, duration: 1, onComplete: () => {
                document.getElementById('screen-1').style.visibility = 'hidden';
                showNarrative(0);
            }});
        }

        function showNarrative(index) {
            if (index < narrativeSteps.length) {
                const textEl = document.getElementById('narrative-text');
                textEl.innerText = narrativeSteps[index];
                
                const screen2 = document.getElementById('screen-2');
                screen2.style.visibility = 'visible';
                
                gsap.fromTo("#screen-2", { opacity: 0, y: 20 }, { opacity: 1, y: 0, duration: 1.5 });
                
                setTimeout(() => {
                    gsap.to("#screen-2", { opacity: 0, y: -20, duration: 1, onComplete: () => {
                        showNarrative(index + 1);
                    }});
                }, 2500);
            } else {
                showFinalScreen();
            }
        }

        function showFinalScreen() {
            const screen3 = document.getElementById('screen-3');
            screen3.style.visibility = 'visible';
            gsap.fromTo("#screen-3", { scale: 0.9, opacity: 0 }, { scale: 1, opacity: 1, duration: 2, ease: "power4.out" });
            
            // Trigger initial light confetti
            confetti({
                particleCount: 100,
                spread: 70,
                origin: { y: 0.6 },
                colors: ['#ffffff', '#e0aaff', '#ff99ac']
            });
        }

        function revealSurprise() {
            // High-end confetti explosion
            const duration = 5 * 1000;
            const animationEnd = Date.now() + duration;
            const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };

            function randomInRange(min, max) {
                return Math.random() * (max - min) + min;
            }

            const interval = setInterval(function() {
                const timeLeft = animationEnd - Date.now();

                if (timeLeft <= 0) {
                    return clearInterval(interval);
                }

                const particleCount = 50 * (timeLeft / duration);
                confetti(Object.assign({}, defaults, { particleCount, origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 } }));
                confetti(Object.assign({}, defaults, { particleCount, origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 } }));
            }, 250);

            // Change content to a personal note
            gsap.to("#final-content", { opacity: 0, duration: 0.5, onComplete: () => {
                document.getElementById('final-content').innerHTML = `
                    <h1 class="playfair text-4xl md:text-5xl mb-6 italic">You are my favorite human.</h1>
                    <p class="text-gray-300 leading-relaxed mb-8">
                        I hope this year brings you as much joy as you give to everyone around you. 
                        Let's make this your best year yet.
                    </p>
                    <div class="text-sm tracking-widest text-white/50">I LOVE YOU</div>
                `;
                gsap.to("#final-content", { opacity: 1, duration: 1 });
            }});
        }
